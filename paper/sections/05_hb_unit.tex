\section{Model hierarki Bayes unit-level: \texttt{hb\_unit}}\label{sec:hbunit}

\subsection{Setting data dan kapan dipakai}\label{sec:hbunit-setting}

Mirip \fct{eblup\_bhf}: dua sumber data --- \code{unit\_data} (mikrodata
survei) dan \code{Xpop} (ringkasan populasi per domain) --- tetapi inferensinya
Bayes penuh lewat \pkg{INLA} (hanya \code{inla\_engine()}). \code{popsize\_var}
bersifat opsional; bila diberikan, prediksi dikombinasikan dengan rata-rata
sampel (koreksi populasi hingga). Dibandingkan \fct{hb\_area}, fungsi ini secara eksplisit
membangkitkan \emph{baris prediksi} untuk setiap domain, sehingga domain tanpa
sampel selalu ikut dievaluasi. Family yang didukung hanya
\code{gaussian}, \code{binomial}, \code{poisson}, dan struktur spasial hanya
\code{none}, \code{bym2}, \code{besag}.

\subsection{Persamaan model}\label{sec:hbunit-model}

\begin{equation}
  y_{ij}=x_{ij}'\beta+u_i+e_{ij},\qquad
  u\sim N\bigl(0,\;\tau_u^{-1}Q_u^{-1}\bigr),\qquad
  e_{ij}\sim\mathcal F,
  \label{eq:hbunit-model}
\end{equation}
dengan $Q_u=\Imat$ (\code{spatial = "none"}), $Q_u=Q_{\text{bym2}}$,
atau $Q_u=Q_{\text{ICAR}}$ (baris 280-294). Tidak ada \code{Ntrials} maupun
\code{E} yang dikirim ke \pkg{INLA} (baris 303-311), sehingga binomial
terbaca sebagai Bernoulli per unit dan Poisson tanpa offset/ekspresi.
Formula acak yang dirangkai (baris 280-294):
\begin{lstlisting}[language=R]
y ~ <fixed_str> + f(..dom_id.., model = 'iid',
                     hyper = list(prec = prior_prec))
\end{lstlisting}

\subsection{Prediktor titik}\label{sec:hbunit-prediktor}

Data INLA disusun dari baris sampel ($y$ nyata) \emph{ditambah satu baris
prediksi per domain dengan} \code{y = NA} (baris 244-278); prediksi dibaca dari
indeks \code{(n\_sample+1):nrow(combined\_df)} (baris 334-342):
\begin{equation}
  \hat\theta_i=\mathbb E[\mu_i\mid y],\qquad
  \mathrm{sd}_i=\sqrt{\Var(\mu_i\mid y)},
  \label{eq:hbunit-pred}
\end{equation}
skala respons (superpopulasi). Bila \code{popsize\_var} diberikan, rerata
domain populasi hingga adalah
$\bar Y_d=f_d\,\bar y_{s,d}+(1-f_d)\bar Y_{d,r}$, $f_d=n_d/N_d$ (dipangkas ke
$[0,1]$), dan baris prediksi dievaluasi pada rata-rata kovariat
\emph{non-sampel} $\bar x_{d,r}=(N_d\bar X_d-n_d\bar x_{s,d})/(N_d-n_d)$,
bukan pada $\bar X_d$ --- sama seperti EBLUP BHF (baris 257-274). Lalu
(baris 350-390)
\begin{equation}
\begin{aligned}
  \hat\theta_d^{\text{adj}}&=f_d\,\bar y_{s,d}+(1-f_d)\,\hat\theta_{d,r},
  \qquad
  \mathrm{CI}=\hat\theta_d^{\text{adj}}\pm1.96\,\mathrm{sd}^{\text{adj}},\\
  \mathrm{sd}^{\text{adj}}&=(1-f_d)\sqrt{\Var(\mu_{d,r}\mid y)+\hat\sigma_e^2/(N_d-n_d)},
\end{aligned}
  \label{eq:hbunit-fpc}
\end{equation}
dengan suku $\hat\sigma_e^2/(N_d-n_d)$ (rerata posterior $\sigma_e^2$) hanya
untuk Gaussian, dan penjagaan batas CI sesuai family (binomial $\to[0,1]$,
Poisson $\ge0$). Untuk domain tanpa sampel $f_d=0$, sehingga
$\hat\theta_d^{\text{adj}}=\hat\theta_d$ --- prediktor murni. Tanpa
\code{popsize\_var}, kuantil posterior INLA dipakai apa adanya (baris
391-397).

\implnote{Pada jalur \code{popsize\_var}, kolom \code{linear\_pred} tetap
memuat $\eta$ baris prediksi tanpa koreksi, sementara \code{hb} sudah
tercampur $f_d\bar y_s+(1-f_d)\hat\theta_{d,r}$, sehingga keduanya tidak lagi
saling invers; sd-nya aproksimasi dari momen, bukan hasil posterior (HB-19).
Jenis interval juga berbeda: kuantil posterior bila tanpa
\code{popsize\_var}, dan normal $\pm1.96$ bila ada (HB-18).}

\subsection{Estimasi komponen varians}\label{sec:hbunit-varians}

\begin{itemize}
  \item Prior presisi komponen acak: \code{prior\_prec} default
        \code{loggamma(0.01, 0.01)} --- \emph{berbeda} dari
        \fct{hb\_area} dan \fct{hb\_twofold} yang memakai
        \code{pc.prec(1, 0.01)} (HB-8).
  \item \code{strategy} default \code{"laplace"}; \code{num.threads} dipaksa
        \code{"1:1"} dengan pemulihan \code{on.exit} (baris 296-301),
        sehingga fungsi ini selalu serial.
  \item \textbf{Start REML untuk pencarian mode.} Prior log-gamma yang samar
        menyisakan punggungan hampir datar menuju $\sigma_u^2=0$ tempat
        pencarian mode INLA dapat berhenti (pada \code{cornsoybean}
        $\sigma_u^2\approx1$ alih-alih $\approx60$, bergantung jumlah
        thread). Untuk Gaussian non-spasial, pencarian mode dimulai dari
        estimasi REML BHF (\code{.bhf\_fit}, bagian~\ref{sec:bhf}):
        \code{control.mode = list(theta = -log(c(}$\hat\sigma_e^2,\hat\sigma_u^2$\code{)), restart = TRUE)}
        (baris 312-318, \code{.hb\_unit\_reml\_start} 539-548), dengan urutan
        hiperparameter INLA presisi likelihood lalu presisi efek acak.
        \code{control.mode} dari \code{inla\_engine(...)} tetap didahulukan.
  \item \code{control.compute} $=$ \code{dic, waic, cpo, mlik} tanpa
        \code{config} (baris 309); argumen \code{inla\_engine(...)} digabung
        lewat \code{modifyList}, sehingga daftar \code{control.*} dapat
        ditimpa per elemen (baris 319-320).
\end{itemize}
Estimasi $\hat\sigma_u^2$ dan, untuk Gaussian, $\hat\sigma_e^2$ adalah rerata
posterior varians $\mathbb E[1/\tau\mid y]$ dari marginal presisi
(\code{.post\_mean\_variance}; baris 415-438); \code{phi} diambil bila ada
(baris 440-444).

\subsection{Estimasi MSE}\label{sec:hbunit-mse}

\begin{equation}
  \widehat{\mathrm{MSE}}=\mathrm{sd}^2,
  \qquad
  \mathrm{RSE}=100\,\mathrm{sd}/\abs{\hat\theta},
  \label{eq:hbunit-mse}
\end{equation}
dengan $\mathrm{sd}$ sudah melalui \eqref{eq:hbunit-fpc} bila
\code{popsize\_var} diberikan (baris 459-460). Varians langsung untuk
keperluan pelaporan diestimasi dari data: Gaussian $\to$
$\hat\sigma_e^2/n_d$, binomial $\to\bar p(1-\bar p)/n$, Poisson
$\to\bar y/n$ (baris 447-456). Tidak ada posterior sampling di fungsi ini.

\subsection{Pseudo-code}\label{sec:hbunit-algo}

\begin{algorithm}[htbp]
\caption{\texttt{hb\_unit} --- alur sebenarnya (\texttt{R/hb\_unit.R:123-548})}
\label{alg:hbunit}
\KwIn{formula, \code{unit\_data}, \code{Xpop}, \code{domain\_var},
  \code{popsize\_var}, \code{family}, \code{spatial}, $W$,
  \code{popnmean\_xpop}, \code{engine}}
\KwOut{objek \code{fastsae\_hb\_unit} berisi \code{df\_hb}, \code{fit}, \code{goodness}}
\code{engine} $\leftarrow$ \texttt{.resolve\_inla\_engine}; hanya INLA (137-150)\;
\texttt{.check\_inla\_installed()};\quad \code{match.arg} (153-156)\;
sampel $\leftarrow$ \texttt{model.frame(na.omit)}; $y_s\leftarrow$ respons (182-190)\;
\code{all\_domains} $\leftarrow$ \texttt{unique(Xpop[[domain\_var]])}\;
\textbf{gagal} bila ada domain sampel tanpa di \code{Xpop} (200-206)\;
\code{unsampled\_domains} $\leftarrow$ \texttt{setdiff(all\_domains, unique(dom\_s))} (209)\;
bila \code{nrow(Xpop)} $>$ jumlah domain: agregasi \code{mean} per domain (211-221)\;
\code{meanxpop} $\leftarrow$ \texttt{model.matrix} atas ringkasan domain (227-235)\;
$W\leftarrow$ \texttt{.convert\_spatial\_weights} (238-242)\;
data INLA $\leftarrow$ baris sampel $+$ \textbf{1 baris \code{y = NA}} per domain, pada $\bar x_{d,r}$ bila \code{popsize\_var} (244-278)\;
formula $\leftarrow$ \code{y \textasciitilde{} <fixed> + f(..dom\_id.., ...)} (280-294)\;
\texttt{num.threads} $\leftarrow$ \code{"1:1"} (dipulihkan via \code{on.exit}) (296-301)\;
\If{Gaussian non-spasial}{\code{control.mode} $\leftarrow$ start REML (312-318)}
\code{fit} $\leftarrow$ \texttt{do.call(INLA::inla, ...)} (322-331)\;
$\hat\theta,\mathrm{sd},\mathrm{CI}\leftarrow$ prediksi indeks \code{(n\_sample+1):n} (333-342)\;
\If{\code{popsize\_var} diberikan}{$f_d\leftarrow$ \eqref{eq:hbunit-fpc};
  $\hat\theta^{\text{adj}},\mathrm{sd}^{\text{adj}},\mathrm{CI}\leftarrow$ (350-390)}
\uElse{CI $\leftarrow$ kuantil INLA (391-397)}\;
$\hat\sigma_u^2,\hat\sigma_e^2\leftarrow\mathbb E[1/\tau\mid y]$; \code{phi} (415-445)\;
$\mathrm{MSE},\mathrm{RSE}\leftarrow$ \eqref{eq:hbunit-mse}; \code{goodness} = list(DIC, WAIC, $mlik$) (458-491)\;
kelas \code{c("fastsae\_hb\_unit","fastsae\_hb","fastsae")} (520)\;
\end{algorithm}

\subsection{Signature, argumen, objek keluaran, dan contoh}\label{sec:hbunit-api}

\begin{lstlisting}[language=R,caption={Signature \texttt{hb\_unit} (\texttt{R/hb\_unit.R:123-136}).}]
hb_unit(formula, unit_data, Xpop, domain_var, popsize_var = NULL,
        family = c("gaussian","binomial","poisson"),
        spatial = c("none","bym2","besag"),
        W = NULL, popnmean_xpop = NULL,
        engine = "inla",   # atau inla_engine(strategy, scale_model,
                           #   prior_prec = loggamma(0.01, 0.01) bila NULL,
                           #   prior_phi = pc(0.5, 0.5) bila NULL, ...)
        print_result = TRUE, ...)  # ...: argumen lama, usang
\end{lstlisting}

Objek keluaran: \code{c("fastsae\_hb\_unit", "fastsae\_hb", "fastsae")} berisi
\code{df\_hb} (\code{domain, y, hb, linear\_pred, vardir, sd, mse, rse,
ci\_lower, ci\_upper, samp\_size, sample\_mean, random\_effect}; ditambah
\code{pop\_size} dan \code{estimated\_total} bila \code{popsize\_var} ada),
\code{estcoef}, \code{hyperpar}, \code{random\_effect\_var},
\code{residual\_var}, \code{phi}, \code{goodness} berupa \code{list(dic,
p\_eff\_dic, waic, p\_eff\_waic, mlik)}, \code{unsampled\_domains},
\code{method}, \code{fit}, \code{engine}, \code{engine\_spec}, \code{call}.

\begin{lstlisting}[language=R,caption={Contoh yang dapat dijalankan (dari \texttt{man/hb\_unit.Rd}).}]
library(fastsae)
data(cornsoybean); data(cornsoybeanmeans)

df_pop <- cornsoybeanmeans
names(df_pop)[names(df_pop) == "MeanCornPixPerSeg"]   <- "CornPix"
names(df_pop)[names(df_pop) == "MeanSoyBeansPixPerSeg"] <- "SoyBeansPix"
df_sample <- cornsoybean
names(df_sample)[names(df_sample) == "County"] <- "CountyIndex"

fit_hb_u <- hb_unit(formula = CornHec ~ CornPix + SoyBeansPix,
                    unit_data = df_sample, Xpop = df_pop,
                    domain_var = "CountyIndex",
                    popsize_var = "PopnSegments")
print(fit_hb_u)
\end{lstlisting}
