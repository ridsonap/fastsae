\section{Model Fay--Herriot spasio-temporal: \texttt{eblup\_stfh}}\label{sec:stfh}

\subsection{Setting data dan kapan dipakai}\label{sec:stfh-setting}

Data berisi satu baris per (domain, waktu) untuk \emph{seluruh} grid
$D\times T$, dengan periode yang sama di setiap domain (pemeriksaan baris,
\code{R/eblup\_stfh.R:284-296}). Respons boleh \code{NA}: sel seperti itu
adalah sel tanpa estimasi langsung, baik periode yang hilang pada domain
tersampel maupun domain yang tidak pernah tersampel. Sel tersebut tidak
dibuang tetapi diprediksi (bagian~\ref{sec:stfh-prediktor}). Variabel bantu
wajib tersedia di setiap sel (baris 262-267), dan \code{vardir} wajib positif
di sel tersampel (baris 274-276). $W$ harus mencakup semua domain, termasuk
yang tidak pernah tersampel. Inti C++ membutuhkan urutan \emph{domain-major};
data diurutkan secara internal (domain menurut kemunculan pertama, yang juga
harus menjadi urutan baris $W$; waktu menaik) dan hasilnya dikembalikan ke
urutan baris pemanggil (baris 204-250, 379-391). Model ini dipakai bila
estimasi tersedia untuk beberapa periode per domain dan kita ingin memakai
kedua jenis korelasi: spasial antar domain dan temporal antar waktu dalam
domain yang sama. Rujukan utamanya \citet{marhuenda2013small};
implementasinya mengikuti \code{sae::eblupSTFH}, yang hanya menerima panel
lengkap.

\subsection{Persamaan model}\label{sec:stfh-model}

\begin{equation}
  y_{dt}=\theta_{dt}+e_{dt},\qquad
  \theta_{dt}=x_{dt}'\beta+u^{(1)}_d+u^{(2)}_{dt},
  \qquad e_{dt}\sim N(0,\psi_{dt}),
  \label{eq:stfh-model}
\end{equation}
dengan dua komponen yang didefinisikan pada \emph{seluruh} grid (penulisan
\emph{domain-major}, $Z=\Imat_D\otimes\mathbf 1_T$):
\begin{align}
  u^{(1)} &\sim N\bigl(0,\;\sigma_1^2\Omega_1\bigr),
    &\Omega_1&=\bigl[(\Imat-\rho_1\Wmat)'(\Imat-\rho_1\Wmat)\bigr]^{-1},
    \label{eq:stfh-spatial}\\
  u^{(2)} &\sim N\bigl(0,\;\Imat_D\otimes\sigma_2^2\Omega_2\bigr),
    &(\Omega_2)_{tt'}&=\frac{\rho_2^{\abs{t-t'}}}{1-\rho_2^2}.
    \label{eq:stfh-temporal}
\end{align}
Hanya sel tersampel $S$ (yaitu $y_{dt}$ tidak \code{NA}) yang mempunyai
$y_{dt}$. Misalkan $S_d$ himpunan periode tersampel domain $d$ dan
$n=\abs S$. Maka
\begin{equation}
  \Vm_S=A+Z_S\,\sigma_1^2\Omega_1Z_S',\qquad
  A=\mathrm{blockdiag}_d\bigl(\sigma_2^2\,\Omega_2[S_d,S_d]+\diag(\psi_d)\bigr),
  \label{eq:stfh-VS}
\end{equation}
dengan blok kosong untuk domain tanpa sampel. Efek spasial \emph{konstan
sepanjang waktu} dalam satu domain, dan $u^{(2)}$ adalah proses AR(1) per
domain. Pada panel lengkap $S$ mencakup seluruh grid dan
\eqref{eq:stfh-VS} menjadi bentuk biasa dengan $M=DT$.

\implnote{Struktur ini \emph{bukan} salah satu interaksi Knorr--Held tipe
I--IV: tidak ada interaksi spasial$\times$temporal di mana efek spasial
berubah tiap waktu. Model \code{model = "S"} mematikan struktur temporal
($\Omega_2=\Imat$, $A$ diagonal). Perbedaan ini tercatat sebagai STFH-1 dan
harus dipahami saat membaca hasil.}

\paragraph{Evaluasi terstruktur, tanpa matriks $n\times n$.} Dengan Woodbury
(\code{src/eblup\_stfh\_core.cpp:104-121})
\begin{equation}
\begin{aligned}
  \Vm^{-1}&=A^{-1}-(A^{-1}Z)\,\mathbf C^{-1}(A^{-1}Z)',
  \qquad
  \mathbf C=(\sigma_1^2\Omega_1)^{-1}+Z'A^{-1}Z,\\
  P&=A^{-1}-U\,\mathrm{blockdiag}(\mathbf C^{-1},Q)\,U',
\end{aligned}
  \label{eq:stfh-woodbury}
\end{equation}
dengan $U=[A^{-1}Z,\;\Vm^{-1}\Xmat]$ ($n\times(D+p)$) dan $Z=Z_S$. Turunan
$\Vm_a$ berbentuk $ZB_aZ'$ ($\sigma_1^2,\rho_1$) atau
$\mathrm{blockdiag}_d(E_a[S_d,S_d])$ ($\sigma_2^2,\rho_2$), sehingga skor,
informasi Fisher, galat baku, dan MSE hanya membutuhkan blok $D\times D$,
$(D+p)\times(D+p)$, dan $\abs{S_d}\times\abs{S_d}$: biaya $O(D^3+nT(D+p))$
per iterasi alih-alih $O(n^3)$, dan memori $O(n(D+p))$ alih-alih $O(n^2)$.
Hasilnya identik secara aljabar dengan perhitungan padat. Log-determinan
memakai lemma determinan
$\log\abs{\Vm}=\log\abs{A}+\log\abs{\sigma_1^2\Omega_1}+\log\abs{\mathbf C}$.
Indeks per domain (baris awal di data ringkas dan periode tersampel) dibangun
oleh \code{st\_make\_index} (baris 83-101). Matriks $T\times T$ yang
diterapkan per blok diiris menjadi $[S_d,S_d]$; pada panel lengkap tidak ada
pengirisan dan aritmetikanya sama persis dengan versi panel lengkap
sebelumnya.

\subsection{Prediktor titik}\label{sec:stfh-prediktor}

Dengan $r=y_S-\Xmat_S\betah$, kedua komponen acak diprediksi dari proyeksi
spasial dan temporalnya (baris 1055-1078):
\begin{equation}
  \hat u^{(1)}=\sigma_1^2\Omega_1\,Z'\Vm^{-1}r,
  \qquad
  \hat u^{(2)}_d=\sigma_2^2\,\Omega_2[\,\cdot\,,S_d]\bigl(\Vm^{-1}r\bigr)_d,
  \qquad
  \hat\theta_{dt}=x_{dt}'\betah+\hat u^{(1)}_d+\hat u^{(2)}_{dt}.
  \label{eq:stfh-pred}
\end{equation}
$\hat u^{(1)}$ identik pada seluruh periode satu domain. Untuk
\code{model = "S"} suku temporal menjadi $\sigma_2^2(\Vm^{-1}r)$ di sel
tersampel.

\paragraph{Sel tanpa estimasi langsung.} \eqref{eq:stfh-pred} berlaku untuk
setiap sel $i=(d,t)$ pada grid, tersampel atau tidak: ia adalah BLUP dari
$\theta_i$ berdasarkan data tersampel \citep{saei2005out}, yaitu
\begin{equation}
  \tilde\theta_i=x_i'\betah+a_i'\Vm^{-1}(y_S-\Xmat_S\betah),\qquad
  a_i=\mathrm{Cov}(\theta_i,y_S)=\sigma_1^2\,\Omega_1[d,d(k)]
  +\mathbf 1\{d(k)=d\}\,\sigma_2^2\,\Omega_2[t,t(k)],
  \label{eq:stfh-blup}
\end{equation}
untuk baris tersampel $k$ dengan domain $d(k)$ dan periode $t(k)$
(\code{st\_full\_u}, baris 672-697; baris 1373-1398). Periode yang hilang
pada domain tersampel meminjam kekuatan dari periode lain domain itu (AR(1))
dan dari tetangganya (SAR). Domain yang tidak pernah tersampel hanya
meminjam dari tetangga, karena $u^{(2)}$ independen antar domain. Ini adalah
\emph{kriging} spasial, seperti pada model SAR Fay--Herriot dengan area di
luar sampel \citep{benedetti2024handling}. Untuk \code{model = "S"}, $u^{(2)}$
sel yang hilang independen dari data sehingga hanya bagian spasial yang
dipinjam. Asumsinya \emph{missing at random}: pola sel tersampel tidak
bergantung pada $\theta$.

\subsection{Estimasi komponen varians}\label{sec:stfh-varians}

Parameter: $\theta=(\sigma_1^2,\rho_1,\sigma_2^2,\rho_2)$ untuk
\code{"ST"} dan $(\sigma_1^2,\rho_1,\sigma_2^2)$ untuk \code{"S"}. Nilai awal
default $0.5\,\mathrm{median}(\psi_S)$ untuk varians dan $0.5$ untuk korelasi
(baris 874-878), semua dapat dioverride lewat argumen \code{*\_start}.
Skoring selalu berbentuk REML (\code{method} di-hardcode \code{"REML"},
\code{R/eblup\_stfh.R:371}) pada sel tersampel, yaitu likelihood marginal
dengan sel yang hilang diintegrasikan keluar:
\begin{equation}
  S_a=-\tfrac12\tr(P V_a)+\tfrac12 y_S'P V_a P y_S,
  \qquad
  F_{ab}=\tfrac12\tr(P V_a P V_b),
  \label{eq:stfh-score}
\end{equation}
dihitung lewat \eqref{eq:stfh-woodbury}. Langkah $F^{-1}S$, $\rho_1,\rho_2$
dipetakan ke $\pm0.999$ bila $\abs\rho\ge1$, konvergensi relatif $\le10^{-4}$
dengan maksimum 100 iterasi (baris 921-953) --- persis iterasi
\code{sae::eblupSTFH}.

Bila iterasi klasik gagal (biasanya karena iterasi varians menjadi negatif
dan $\Vm$ tak-definit saat $\rho_1\to1$) atau konvergen ke varians negatif,
dijalankan \emph{fallback} REML terkendala \code{stfh\_fit\_safeguarded}
(baris 557-668, 955-984): Fisher scoring terproyeksi pada $\sigma^2_{1,2}\ge0$,
$\abs{\rho}\le0.999$, parameter di batas aktif ditahan (termasuk korelasi dari
varians yang bernilai $0$, yang tidak teridentifikasi), dan \emph{step
halving} sampai log-likelihood REML tidak turun. Fit yang konvergen pada
iterasi klasik tidak berubah.

Galat baku $\beta$ dari $\sqrt{\mathrm{diag}(Q)}$; nilai $p$ dari distribusi
normal (kolom \code{tvalue} disalin ke \code{zvalue} di R, STFH-6). Galat
baku komponen varians mengikuti \code{sae::eblupSTFH} (invers informasi
iterasi klasik terakhir, STFH-7) bila iterasi itu konvergen, dan dari
informasi di estimasi akhir bila fallback dipakai; parameter yang tidak
teridentifikasi di batas mendapat \code{NA} (baris 1120-1158).

\subsection{Estimasi MSE}\label{sec:stfh-mse}

Kolom \code{mse} dan \code{rse} diisi \code{NA} kecuali
\code{compute\_mse = TRUE} (\code{R/eblup\_stfh.R:402-444}; STFH-2). Ada dua
jenis lewat \code{mse\_type}:

\paragraph{Analitik (default), sel tersampel.} MSE Prasad--Rao orde kedua
\citep{marhuenda2013small} dalam bentuk terstruktur (baris 1161-1370):
\begin{align}
  g_1&=\psi-\psi^2\diag(\Vm^{-1}),\qquad
  g_2=\psi^2\diag\bigl(\Vm^{-1}\Xmat Q\Xmat'\Vm^{-1}\bigr),
  \label{eq:stfh-g12}\\
  g_3(i)&=\psi_i^2\sum_{a,b}F^{ab}\,(PV_aPV_bP)_{ii},\qquad
  g_4(i)=\tfrac12\sum_{j,k}F^{jk}\psi_i^2\,(PV_{jk}P)_{ii},
  \label{eq:stfh-g34}
\end{align}
$\widehat{\mathrm{MSE}}=g_1+g_2+2g_3-g_4$. Bentuk $g_3$ diturunkan dari
$\tilde\theta(\theta)=y-\Psi P(\theta)y$; ia setara dengan bentuk buku teks
$\psi_i^2\sum F^{ab}(\Vm^{-1}V_a\Vm^{-1}\Vm\Vm^{-1}V_b\Vm^{-1})_{ii}$ sampai
$o(m^{-1})$, tetapi tidak terpengaruh komponen $V_a$ yang berada di ruang
kolom $\Xmat$ (yang meledak pada bentuk buku teks saat $\rho_1\to1$ dengan $W$
row-standar). $g_4$ menangkap kelengkungan $\Vm$ dalam $\rho_1,\rho_2$ (turunan
kedua $V_{jk}$). Korelasi di batas ($\abs\rho\ge0.999$) diperlakukan tetap,
seperti pada SFH. Komponen $g_1$--$g_4$ juga dikembalikan sebagai elemen
objek.

\paragraph{Analitik, sel tanpa estimasi langsung.} Untuk prediktor
\eqref{eq:stfh-blup} dengan $R_i=x_i'-a_i'\Vm^{-1}\Xmat$ (baris 1396-1500):
\begin{equation}
\begin{aligned}
  g_1(i)&=\mathrm{Var}(\theta_i-x_i'\beta)-a_i'\Vm^{-1}a_i,\qquad
  g_2(i)=R_iQR_i',\\
  g_3(i)&=\sum_{a,b}F^{ab}\,\ell_a'P\ell_b,\qquad
  \ell_a'=\partial_a a_i'-\bigl(a_i'\Vm^{-1}+R_iQ\Xmat'\Vm^{-1}\bigr)V_a,\\
  \widehat{\mathrm{MSE}}(i)&=g_1+g_2+g_3,
\end{aligned}
  \label{eq:stfh-mse-oos}
\end{equation}
dengan $\mathrm{Var}(\theta_i-x_i'\beta)=\sigma_1^2\Omega_1[d,d]+\sigma_2^2\Omega_2[t,t]$
(atau $+\sigma_2^2$ untuk \code{"S"}). $g_3$ berasal dari turunan eksak
prediktor terhadap komponen varians, $\partial_a\tilde\theta_i=\ell_a'Py$.
Kolom \code{g4} bernilai $0$ untuk sel ini.

\implnote{Koreksi orde kedua untuk bias $g_1(\hat\theta)$, yaitu
$\tfrac12\tr(F^{-1}\nabla^2g_1)$ (dipakai untuk domain tak-tersampel pada
\fct{eblup\_sfh}), sempat diimplementasikan dan cocok dengan turunan beda hingga
dari implementasi padat. Namun koreksi itu tidak stabil secara numerik bila
$\rho_1$ mendekati $\pm1$ ($\Omega_1$ hampir singular) atau bila suatu
komponen varians lemah teridentifikasi: dalam simulasi
(Tabel~\ref{tab:stfh-unsampled-sim}) beberapa replikasi menghasilkan suku
yang meledak sehingga bias relatif rata-ratanya ratusan hingga ribuan persen.
Tanpa koreksi itu, $g_1+g_2+g_3$ memberi bias relatif beberapa persen dan
cakupan mendekati 95\% untuk kedua jenis sel tanpa sampel. Karena itu koreksi
tidak dipakai.}

\paragraph{Bootstrap parametrik} (\code{mse\_type = "bootstrap"},
\code{src/eblup\_stfh\_pbmse.cpp}):
\begin{enumerate}[label=\arabic*.]
  \item fit awal penuh pada data asli;
  \item bangkitkan $u^{(1)\ast}\sim N(0,\sigma_1^2\Omega_1)$ pada semua $D$
        domain lewat faktor Cholesky \emph{bawah} $\Omega_1=LL'$, $u^{(2)\ast}$
        AR(1) per domain pada semua $T$ periode, dan $\varepsilon^\ast\sim
        N(0,\psi)$ hanya pada sel tersampel --- \emph{sekuensial, sebelum}
        region paralel (baris 158-188);
  \item dalam \texttt{\#pragma omp parallel for schedule(dynamic)}:
        $y_S^\ast=\mu^\ast_S+\varepsilon^\ast$ dengan
        $\mu^\ast=\Xmat\hat\beta+Zu^{(1)\ast}+u^{(2)\ast}$, lalu komponen
        varians \emph{diestimasi ulang} dengan REML pada sel tersampel
        (\code{stfh\_fit\_arma}) seperti \code{sae::pbmseSTFH}, dan setiap
        sel grid diprediksi dengan \eqref{eq:stfh-blup};
  \item replikat yang fit-nya gagal diganti draw baru sampai tepat $B$
        replikat valid (maksimum $5B$ percobaan);
        $\widehat{\mathrm{MSE}}_i=B^{-1}\sum_b(\hat\theta_{b,i}^\ast-\mu_{b,i}^\ast)^2$
        untuk \emph{setiap} sel, tersampel atau tidak.
\end{enumerate}
Bila \code{mse\_type} tidak diberikan tetapi \code{B} diberikan, bootstrap
yang dipilih (kompatibilitas dengan versi sebelumnya, baris 190-198).

\paragraph{Validasi.} Prediksi semua sel, $\betah$, $g_1$, dan $g_2$
dibandingkan dengan implementasi padat (pembentukan $\Vm_S$ penuh) dan
cocok sampai presisi mesin. $g_3$ cocok sampai $10^{-10}$ relatif, dan
$g_4$ sel tersampel cocok dengan turunan beda hingga
(\code{tests/testthat/test-eblup\_stfh\_unsampled.R}). Pada panel lengkap,
hasilnya identik bit demi bit dengan versi sebelum dukungan sel tak tersampel.
Tabel~\ref{tab:stfh-unsampled-sim} merangkum simulasi Monte Carlo berbasis
model.

\subsection{Signature, argumen, objek keluaran, dan contoh}\label{sec:stfh-api}

\begin{lstlisting}[language=R,caption={Signature \texttt{eblup\_stfh} (\texttt{R/eblup\_stfh.R:164-188}).}]
eblup_stfh(formula, vardir, data, domain, time, W,
           model = c("ST", "S"), maxiter = 100, precision = 1e-4,
           sigma21_start = NULL, rho1_start = 0.5,
           sigma22_start = NULL, rho2_start = 0.5,
           compute_mse = FALSE,
           mse_type = c("analytical", "bootstrap"),
           B = 100, n_threads = 1, seed = -1,
           self_benchmark = FALSE, benchmark_weight = NULL,
           benchmark_target = NULL, benchmark_group = NULL,
           print_result = TRUE)
\end{lstlisting}

\begin{itemize}
  \item \code{domain}, \code{time} --- nama kolom/kolom formula penanda domain
        dan waktu (wajib; satu baris per sel grid, urutan baris bebas).
  \item \code{W} --- matriks tetangga $D\times D$ (hanya antar domain, tanpa
        dimensi waktu) untuk \emph{semua} domain, dengan urutan domain menurut
        kemunculan pertama.
  \item \code{model} --- \code{"ST"} (default, AR(1) temporal) atau
        \code{"S"} (tanpa struktur temporal).
  \item \code{compute\_mse}, \code{mse\_type} --- \code{FALSE} (default)
        menghasilkan \code{mse = NA}; \code{TRUE} dengan \code{"analytical"}
        (default) atau \code{"bootstrap"}.
  \item \code{self\_benchmark} --- kalibrasi agregat berbobot sel tersampel
        per kelompok (\code{.setup\_self\_benchmark} menormalkan bobot pada
        sel tersampel).
\end{itemize}

Objek keluaran: \code{"fastsae"} berisi \code{estcoef}
(\code{beta, std.error, tvalue, pvalue, stderr\_beta, zvalue}),
\code{estvarcomp} (4 atau 3 baris \code{estimate, std.error}), \code{goodness}
(\code{loglikelihood, AIC, BIC} dari log-likelihood penuh seperti
\pkg{sae::eblupSTFH}, plus \code{loglikelihood\_reml}; sebelum 1.0.0 elemen
pertama bernama \code{loglike}, STFH-8), \code{df\_eblup}
dengan kolom \code{domain, time, y, eblup, vardir, mse, rse, mse\_pb,
random\_effect\_u1, random\_effect\_u2, sampled} (\code{y} dan \code{vardir}
\code{NA} di sel tanpa estimasi langsung), \code{mse\_analytical},
\code{rse\_analytical}, \code{g1}--\code{g4}, \code{mse\_type}, \code{B},
\code{convergence}, \code{n\_iter}, \code{method = "REML"}.

\begin{lstlisting}[language=R,caption={Contoh yang dapat dijalankan (dari \texttt{man/eblup\_stfh.Rd}); domain 21 dan 25 tidak pernah tersampel.}]

\subsection{Referensi kunci}\label{sec:stfh-ref}

\citet{marhuenda2013small} adalah rujukan model dan sekaligus rujukan yang
disebut roxygen (\code{R/eblup\_stfh.R}) bersama \citet{rao2015small};
gabungan data deret waktu dengan potongan lintas bagian yang menjadi dasar
model semacam ini dirintis \citet{rao1994small}. Prediksi sel di luar sampel
mengikuti \citet{saei2005out} dan, untuk efek SAR pada area di luar sampel,
\citet{benedetti2024handling}. Implementasi pembandingnya \pkg{sae}
\citep{molina2015sae}. Seluruh rujukan terdaftar di \code{references.bib}.
